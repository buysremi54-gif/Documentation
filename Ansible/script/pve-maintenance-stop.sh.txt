#!/bin/bash

set -Eeuo pipefail

# ==============================
# Configuration
# ==============================

LOGFILE="/var/log/pve-maintenance.log"
VM_LIST="/var/log/pve-maintenance-vms.list"

SERVICES=(
    corosync
    pve-cluster
    pveproxy
    pvedaemon
    pvestatd
    pve-ha-lrm
    pve-ha-crm
)

exec > >(tee -a "$LOGFILE") 2>&1

trap 'echo "[ERREUR] ligne $LINENO : $BASH_COMMAND"' ERR


# ==============================
# Fonctions
# ==============================

error()
{
    echo "[ERREUR] $1"
    exit 1
}


# retry <essais> <délai en s> <description> <commande...>
retry()
{
    local max=$1 delay=$2 desc=$3 n=1
    shift 3

    until "$@"
    do
        [ "$n" -ge "$max" ] && error "$desc : échec après $max tentatives"
        echo "$desc : tentative $n/$max échouée, nouvel essai dans ${delay}s"
        sleep "$delay"
        n=$((n + 1))
    done

    echo "$desc : OK"
}


pmxcfs_ok()
{
    mountpoint -q /etc/pve && systemctl is-active --quiet pve-cluster
}


services_ok()
{
    local S

    for S in "${SERVICES[@]}"
    do
        systemctl is-active --quiet "$S"  || { echo "[ERREUR] $S arrêté"; return 1; }
        systemctl is-enabled --quiet "$S" || { echo "[ERREUR] $S non activé au démarrage"; return 1; }
    done

    return 0
}


quorum_ok()
{
    pvecm status | grep -q '^Quorate:.*Yes'
}


ceph_ok()
{
    # Ceph non configuré sur ce cluster : on ignore
    [ -f /etc/pve/ceph.conf ] || return 0

    local s
    s=$(timeout 20 ceph health 2>/dev/null | awk '{print $1}') || return 1

    echo "Ceph : $s"
    [ -n "$s" ] && [ "$s" != "HEALTH_ERR" ]
}


in_cluster()
{
    local out
    out=$(pvecm nodes) || return 1

    # grep -w sur la sortie capturée : pas de pipe, pas de SIGPIPE
    grep -qw "$NODE" <<< "$out"
}


maintenance_off()
{
    ha-manager crm-command node-maintenance disable "$NODE"
}


# IPv4 de la VM : interface qui porte la MAC de net0 (ignore docker0, bridges, IPv6)
vm_ip()
{
    local mac

    mac=$(qm config "$1" | grep -oiP '^net0: [^=]+=\K([0-9a-f]{2}:){5}[0-9a-f]{2}') || return 1

    qm guest cmd "$1" network-get-interfaces 2>/dev/null | MAC="${mac,,}" perl -MJSON::PP -0777 -ne '
        for my $i (@{ decode_json($_) }) {
            next if lc($i->{"hardware-address"} // "") ne $ENV{MAC};
            for my $a (@{ $i->{"ip-addresses"} // [] }) {
                if ($a->{"ip-address-type"} eq "ipv4") { print $a->{"ip-address"}; exit }
            }
        }' 2>/dev/null | grep .
}


# Attend le SSH (port 22) d'une VM (5 min max) et dit pourquoi ça échoue
wait_vm_ssh()
{
    local vmid=$1 ip n

    for n in $(seq 30)
    do
        ip=$(vm_ip "$vmid" || true)

        if [ -z "$ip" ]
        then
            echo "VM $vmid : IP introuvable via l'agent ($n/30)"
        elif timeout 5 bash -c 'exec 3<>/dev/tcp/$0/22' "$ip" 2>/dev/null
        then
            echo "VM $vmid ($ip) : SSH OK"
            return 0
        else
            echo "VM $vmid ($ip) : port 22 fermé ($n/30)"
        fi

        sleep 10
    done

    return 1
}


start_vms()
{
    echo "Redémarrage des VM arrêtées pendant la maintenance..."

    if [ ! -f "$VM_LIST" ]
    then
        echo "Aucune VM à redémarrer"
        return 0
    fi

    local VMID

    while read -r VMID || [ -n "$VMID" ]
    do
        [ -n "$VMID" ] || continue

        # VM déjà démarrée (par HA par exemple) : on passe
        if qm status "$VMID" 2>/dev/null | grep -q "running"
        then
            echo "VM $VMID déjà démarrée"
            wait_vm_ssh "$VMID" || error "VM $VMID : SSH KO, arrêt avant les VM suivantes"
            continue
        fi

        echo "Démarrage VM $VMID"
        if qm start "$VMID"
        then
            wait_vm_ssh "$VMID" || error "VM $VMID : SSH KO, arrêt avant les VM suivantes"
        else
            echo "[WARN] VM $VMID non démarrée"
        fi

    done < "$VM_LIST"

    rm -f "$VM_LIST"

    echo "Redémarrage des VM terminé"
}


# ==============================
# Programme principal
# ==============================

echo "================================="
echo "Fin maintenance PVE"
date
echo "================================="

NODE=$(hostname -s)
echo "Noeud traité : $NODE"

# /etc/pve (pmxcfs) doit être monté après le reboot
retry 30 10 "Montage /etc/pve"        pmxcfs_ok

retry 10 30 "Services PVE"            services_ok
retry 30 10 "Quorum"                  quorum_ok
retry 20 15 "Ceph"                    ceph_ok
retry 10 10 "Présence de $NODE"       in_cluster

retry 12 10 "Sortie maintenance HA"   maintenance_off

start_vms

echo "$NODE terminé"

echo
echo "================================="
echo "Maintenance cluster terminée"
echo "================================="
